Creates a note and returns the stored note with its identifier. Use the **Try it** button to send a test request from your browser.

<ApiEndpoint method="POST" path="/v1/notes" baseUrl="https://api.example.com/v1">

<ApiRequestExample client="curl" lang="bash" label="Create a note">

```bash
curl -X POST "https://api.example.com/v1/notes" \
  -H "Authorization: Bearer sk_test_123" \
  -H "Content-Type: application/json" \
  -d '{"body":"Ship the docs today."}'
```

</ApiRequestExample>

<ApiRequestExample client="node-fetch" lang="js" label="Create a note">

```js
const response = await fetch("https://api.example.com/v1/notes", {
  method: "POST",
  headers: {
    Authorization: "Bearer sk_test_123",
    "Content-Type": "application/json",
  },
  body: JSON.stringify({ body: "Ship the docs today." }),
});
```

</ApiRequestExample>

<ApiResponseExample status="201" contentType="application/json">

```json
{ "id": "note_123", "body": "Ship the docs today.", "createdAt": "2026-01-01T12:00:00Z" }
```

</ApiResponseExample>

<ApiResponseExample status="400" contentType="application/json">

```json
{ "code": "invalid_body", "message": "The body field is required." }
```

</ApiResponseExample>

<ApiParams title="Body" detail="application/json">
  <ApiParam name="body" type="string" location="body" required>
    The text of the note.
  </ApiParam>
</ApiParams>

<ApiParams title="Response" detail="application/json">
  <ApiParam name="id" type="string" location="response" required>
    The identifier of the note.
  </ApiParam>
  <ApiParam name="body" type="string" location="response" required>
    The text of the note.
  </ApiParam>
  <ApiParam name="createdAt" type="string" location="response" required>
    The time that the note was created, in ISO 8601 format.
  </ApiParam>
</ApiParams>

</ApiEndpoint>
